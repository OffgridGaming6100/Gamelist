// DOM Handles (continued) const selectedCountEl = document.getElementById('selectedCount'); const btnClearAll = document.getElementById('btnClearAll'); const targetDriveSelect = document.getElementById('targetDriveSelect'); const capacityBarFill = document.getElementById('capacityBarFill'); const btnCopyDrawer = document.getElementById('btnCopyDrawer'); const searchInput = document.getElementById('searchInput'); const sortSelect = document.getElementById('sortSelect'); const genreChips = document.getElementById('genreChips'); const rigPresets = document.getElementById('rigPresets'); // Modal DOM const reqsModal = document.getElementById('reqsModal'); const btnCloseModal = document.getElementById('btnCloseModal'); const modalGameTitle = document.getElementById('modalGameTitle'); const modalRam = document.getElementById('modalRam'); const modalVram = document.getElementById('modalVram'); const modalCpu = document.getElementById('modalCpu'); const modalGpu = document.getElementById('modalGpu'); // Initial Initialization document.addEventListener('DOMContentLoaded', () => { updateRigCompatibility(); attachEventListeners(); }); function attachEventListeners() { // Preset Rig Selection rigPresets.addEventListener('click', (e) => { const btn = e.target.closest('.btn-preset'); if (!btn) return; rigPresets.querySelectorAll('.btn-preset').forEach(b => b.classList.remove('active')); btn.classList.add('active'); currentRig.ram = parseInt(btn.dataset.ram); currentRig.vram = parseInt(btn.dataset.vram); updateRigCompatibility(); }); // Grid Action Click Handling gameGrid.addEventListener('click', (e) => { const btnPlay = e.target.closest('.btn-play'); const btnReqs = e.target.closest('.btn-reqs'); if (btnPlay) { const card = btnPlay.closest('.game-card'); toggleGameSelection(card); } else if (btnReqs) { const card = btnReqs.closest('.game-card'); openReqsModal(card); } }); // Genre Chips Filter genreChips.addEventListener('click', (e) => { const chip = e.target.closest('.chip'); if (!chip) return; genreChips.querySelectorAll('.chip').forEach(c => c.classList.remove('active')); chip.classList.add('active'); activeGenre = chip.dataset.genre; filterAndSortGames(); }); // Search & Sort Inputs searchInput.addEventListener('input', filterAndSortGames); sortSelect.addEventListener('change', filterAndSortGames); // Storage Target Drive Switcher targetDriveSelect.addEventListener('change', updateCartUI); // Cart Actions btnClearAll.addEventListener('click', clearAllSelection); btnCopyDrawer.addEventListener('click', copyPayloadList); // Cart Item Removal (Delegated) cartItems.addEventListener('click', (e) => { const removeBtn = e.target.closest('.btn-remove-item'); if (!removeBtn) return; const gameId = removeBtn.dataset.id; const card = gameGrid.querySelector(`.game-card[data-id="${gameId}"]`); if (card) toggleGameSelection(card); }); // Modal Close Events btnCloseModal.addEventListener('click', closeModal); window.addEventListener('click', (e) => { if (e.target === reqsModal) closeModal(); }); window.addEventListener('keydown', (e) => { if (e.key === 'Escape') closeModal(); }); } // --- RIG ANALYSIS & HUD LOGIC --- function updateRigCompatibility() { const cards = gameGrid.querySelectorAll('.game-card'); cards.forEach(card => { const reqRam = parseInt(card.dataset.ram); const reqVram = parseInt(card.dataset.vram); const id = card.dataset.id; const hud = document.getElementById(`hud-${id}`); if (!hud) return; if (currentRig.ram >= reqRam && currentRig.vram >= reqVram) { hud.className = 'hud-status-badge status-ready'; hud.innerHTML = ' RIG READY'; } else if (currentRig.ram < reqRam && currentRig.vram < reqVram) { hud.className = 'hud-status-badge status-warning'; hud.innerHTML = ' RAM & VRAM LOW'; } else if (currentRig.ram < reqRam) { hud.className = 'hud-status-badge status-warning'; hud.innerHTML = ' RAM LOW'; } else { hud.className = 'hud-status-badge status-vram'; hud.innerHTML = ' VRAM LOW'; } }); } // --- SELECTION & CART LOGIC --- function toggleGameSelection(card) { const id = card.dataset.id; const title = card.dataset.title; const size = parseInt(card.dataset.size); const btnPlay = card.querySelector('.btn-play'); if (selectedGames.has(id)) { selectedGames.delete(id); card.classList.remove('selected'); btnPlay.innerHTML = ' Select Game'; showToast(`Removed ${title}`); } else { selectedGames.set(id, { id, title, size }); card.classList.add('selected'); btnPlay.innerHTML = ' Selected'; showToast(`Added ${title}`); } updateCartUI(); } function clearAllSelection() { selectedGames.clear(); const cards = gameGrid.querySelectorAll('.game-card'); cards.forEach(card => { card.classList.remove('selected'); const btnPlay = card.querySelector('.btn-play'); btnPlay.innerHTML = ' Select Game'; }); updateCartUI(); showToast('Cleared all selected games'); } function updateCartUI() { const count = selectedGames.size; selectedCountEl.textContent = `(${count})`; btnClearAll.style.display = count > 0 ? 'block' : 'none'; btnCopyDrawer.disabled = count === 0; if (count === 0) { cartItems.innerHTML = '
No games selected.
'; } else { cartItems.innerHTML = Array.from(selectedGames.values()).map(game => `
  • ${game.title}
    ${game.size} GB
    • `).join(''); } // Calculate HDD Storage Fill let usedGB = 0; selectedGames.forEach(g => usedGB += g.size); const driveLimitGB = parseInt(targetDriveSelect.value); totalSizeEl.textContent = `${usedGB} / ${driveLimitGB} GB`; const fillPercentage = Math.min((usedGB / driveLimitGB) * 100, 100); capacityBarFill.style.width = `${fillPercentage}%`; if (usedGB > driveLimitGB) { capacityBarFill.classList.add('overload'); totalSizeEl.style.color = 'var(--accent-red)'; } else { capacityBarFill.classList.remove('overload'); totalSizeEl.style.color = 'var(--accent-orange)'; } } // --- SEARCH, SORT & FILTER --- function filterAndSortGames() { const query = searchInput.value.toLowerCase().trim(); const sortVal = sortSelect.value; const cards = Array.from(gameGrid.querySelectorAll('.game-card')); // Filter cards.forEach(card => { const title = card.dataset.title.toLowerCase(); const genre = card.dataset.genre; const matchesSearch = title.includes(query); const matchesGenre = (activeGenre === 'all' || genre === activeGenre); card.style.display = (matchesSearch && matchesGenre) ? 'flex' : 'none'; }); // Sort cards.sort((a, b) => { if (sortVal === 'name-asc') { return a.dataset.title.localeCompare(b.dataset.title); } else if (sortVal === 'size-asc') { return parseInt(a.dataset.size) - parseInt(b.dataset.size); } else if (sortVal === 'size-desc') { return parseInt(b.dataset.size) - parseInt(a.dataset.size); } return 0; }); cards.forEach(card => gameGrid.appendChild(card)); } // --- MODAL UTILS --- function openReqsModal(card) { modalGameTitle.textContent = card.dataset.title; modalRam.textContent = `${card.dataset.ram} GB`; modalVram.textContent = `${card.dataset.vram} GB`; modalCpu.textContent = card.dataset.cpu; modalGpu.textContent = card.dataset.gpu; reqsModal.classList.add('active'); } function closeModal() { reqsModal.classList.remove('active'); } // --- UTILITY: COPY TO CLIPBOARD --- function copyPayloadList() { if (selectedGames.size === 0) return; let totalGB = 0; let textOutput = `=== OFFGRID GAMING SELECTION LIST ===\n`; selectedGames.forEach(game => { textOutput += `- ${game.title} (${game.size} GB)\n`; totalGB += game.size; }); const driveLabel = targetDriveSelect.options[targetDriveSelect.selectedIndex].dataset.label; textOutput += `\nTotal Payload Size: ${totalGB} GB\nTarget Disk: ${driveLabel}`; navigator.clipboard.writeText(textOutput).then(() => { showToast('Selection list copied to clipboard!'); }).catch(() => { showToast('Failed to copy list.'); }); } // --- UTILITY: TOAST NOTIFICATIONS --- function showToast(message) { const container = document.getElementById('toastContainer'); const toast = document.createElement('div'); toast.className = 'toast'; toast.innerHTML = ` ${message}`; container.appendChild(toast); setTimeout(() => { toast.remove(); }, 2500); }
